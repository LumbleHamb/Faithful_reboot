public class ProductionBuilding : MonoBehaviour {
    public BuildingData data;
    public ResourceType outputType;
    public int productionRatePerHour = 50;
    public int maxCapacity = 150;
    public float currentStoredResource = 55f;
    public string assignedWorkerName = "Destiny Shaw";
    public string assignedWorkerRole = "Wood Cutter";

    void Update() {
        if (currentStoredResource < maxCapacity) {
            currentStoredResource += (productionRatePerHour / 3600f) * Time.deltaTime;
            currentStoredResource = Mathf.Min(currentStoredResource, maxCapacity);
        }
    }

    public void CollectResource(PlayerAccount player) {
        int amountToCollect = Mathf.FloorToInt(currentStoredResource);
        player.AddResource(outputType, amountToCollect);
        currentStoredResource -= amountToCollect;
    }
}
